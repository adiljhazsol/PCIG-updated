\section{Introduction}

This document provides comprehensive API specifications and functionality requirements for the PCIG (Property Capital Investment Group) platform backend. The platform manages tax deed/lien investments with two main user roles: \textbf{Admin/Workers} and \textbf{Investors}.

\subsection{Technology Stack}
\begin{itemize}
    \item \textbf{Backend Framework:} Laravel 10+
    \item \textbf{Database:} MySQL/PostgreSQL
    \item \textbf{Authentication:} Laravel Sanctum (API tokens)
    \item \textbf{Queue System:} Redis/Database queues
    \item \textbf{Storage:} AWS S3 or local storage
\end{itemize}

\subsection{Core Data Models}
\begin{itemize}
    \item Users (Admin, Worker, Investor roles)
    \item Properties (with workflow stages)
    \item Funds
    \item Transactions
    \item Documents
    \item Expenses \& Allocations
    \item Redemptions \& Payoffs
\end{itemize}

\begin{itemize}

    \item Extract text using OCR (Tesseract/AWS Textract)

\subsubsection{Calculation Methods}
\begin{lstlisting}[language=php]
// Simple Interest
$interest = $principal * ($rate / 100) * $days / 365;

// Compound Interest (daily)
$interest = $principal * (pow(1 + $rate/365, $days) - 1);

// Georgia Redemption Interest (20% simple)
$interest = $bidPrice * 0.20 * $days / 365;
\end{lstlisting}
